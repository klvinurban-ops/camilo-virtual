function mediaHTML(p){
  const id1=p.img?driveId(p.img):'';
  const id2=p.img2?driveId(p.img2):'';
  const bannerCls=p.ph==='mini'?'ph mini':'ph';
  const bannerTxt=p.ph==='mini'?'Mini LED':p.marca;
  let html='<div class="img-wrap">';
  if(id1){
    const s=driveSrcs(id1);
    html+=`<img src="${s[0]}" alt="${esc(p.nombre)}" class="img-main" id="main-${p.id}" loading="lazy" data-alts='${JSON.stringify(s.slice(1))}' data-banner="ban-${p.id}" onerror="imgFail(this)">`;
    html+=`<div class="${bannerCls}" id="ban-${p.id}" style="display:none"><b>${bannerTxt}</b></div>`;
  }else{
    html+=`<div class="${bannerCls}" id="ban-${p.id}"><b>${bannerTxt}</b></div>`;
  }
  if(id2){
    const s2=driveSrcs(id2,'w300');
    html+=`<img src="${s2[0]}" alt="Interior" class="img-thumb" id="thumb-${p.id}" loading="lazy" data-alts='${JSON.stringify(s2.slice(1))}' onerror="imgFailThumb(this)" onclick="swapImg('${p.id}')">`;
  }
  html+='</div>';
  return html;
}

  const tS=main.src,tA=main.getAttribute('data-alts');
  main.src=thumb.src;main.setAttribute('data-alts',thumb.getAttribute('data-alts')||'[]');
  thumb.src=tS;thumb.setAttribute('data-alts',tA||'[]');
  main.style.display='block';
  const ban=$('ban-'+pid);if(ban)ban.style.display='none';
}

  $('stepTag').textContent=`Paso ${idx+1} de ${total}`;
  $('btnBack').style.display=idx>0?'block':'none';
  $('btnNext').textContent=(idx===total-1)?'📤 Radicar caso':'Siguiente →';
  if(idx===total-1)buildSummary();
  $('btnNext').dataset.idx=idx; $('btnBack').dataset.idx=idx;
  saveDraft();
}
function nextStep(){
  const idx=parseInt($('btnNext').dataset.idx||'0',10);
  if(!validateStep(curStep))return;
  if(idx===flujoActual.length-1){submitGar();return;}
  showStepByIndex(idx+1);
}
function prevStep(){
  const idx=parseInt($('btnBack').dataset.idx||'0',10);
  if(idx>0)showStepByIndex(idx-1);
}
function markErr(inputId,errId,bad){
  $(errId).style.display=bad?'block':'none';
  if(inputId)$(inputId).classList.toggle('bad',bad);
  return !bad;
}
function validateStep(step){
  if(step===1){if(!gTipo){$('err-tipo').style.display='block';return false;}return true;}
  if(step===2){
    let ok=true;
    ok=markErr('g-nombre','e-nombre',!$('g-nombre').value.trim())&&ok;
    ok=markErr('g-cc','e-cc',!/^[0-9]{6,12}$/.test($('g-cc').value.trim()))&&ok;
    ok=markErr('g-cel','e-cel',!/^3[0-9]{9}$/.test($('g-cel').value.trim()))&&ok;
    ok=markErr(null,'e-mun-dir',!$('g-mun').value.trim()||!$('g-dir').value.trim())&&ok;
    ok=markErr('g-mail','e-mail',!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test($('g-mail').value.trim()))&&ok;
    return ok;
  }
  if(step===3){
    let ok=true;
    ok=markErr(null,'e-marca-serial',!$('g-marca').value.trim()||!$('g-serial').value.trim())&&ok;
    ok=markErr('g-fecha','e-fecha',!$('g-fecha').value)&&ok;
    ok=markErr('g-dano','e-dano',!$('g-dano').value)&&ok;
    return ok;
  }
  if(step===4){
    const ok=files.serial&&files.novedad&&files.video;
    $('e-evid').style.display=ok?'none':'block';
    return ok;
  }
  if(step===5){return markErr('g-nit','e-nit',!/^[0-9-]{6,15}$/.test($('g-nit').value.trim()));}
  return true;
}

}

function buildSummary(){
  const r=[['Tipo de garantía',gTipo],['Marca',$('g-marca').value],['Serial',$('g-serial').value],
  ['Fecha de compra',$('g-fecha').value],['Tipo de daño',$('g-dano').value],
  ['Descripción',$('g-desc').value||'—'],['NIT Corbeta',$('g-nit').value],
  ['Evidencias','Foto serial ✓ · Foto novedad ✓ · Video ✓']];
  if(gTipo==='Postventa'){
    r.splice(1,0,['Cliente final',$('g-nombre').value],['CC',$('g-cc').value],['Celular',$('g-cel').value],
    ['Municipio',$('g-mun').value],['Dirección',$('g-dir').value],['Correo',$('g-mail').value]);
  }
  $('sumBox').innerHTML='<div class="sum-h">Resumen del caso</div>'+
    r.map(([k,v])=>`<div class="sum-r"><b>${k}</b><span>${v}</span></div>`).join('');
}
function submitGar(){
  const total=(files.serial.size+files.novedad.size+files.video.size)/1048576;
  if(total>35){alert('⚠️ Las evidencias pesan '+total.toFixed(0)+' MB (máximo 35).');return;}
  const caso={tipo:'GARANTÍA · '+gTipo,fecha:new Date().toLocaleString('es-CO'),
    marca:$('g-marca').value,serial:$('g-serial').value,fechaCompra:$('g-fecha').value,
    dano:$('g-dano').value,desc:$('g-desc').value,nit:$('g-nit').value};
  if(gTipo==='Postventa'){
    caso.nombre=$('g-nombre').value;caso.cc=$('g-cc').value;caso.cel=$('g-cel').value;
    caso.municipio=$('g-mun').value;caso.direccion=$('g-dir').value;caso.correo=$('g-mail').value;
  }
  procesarCaso(caso,'¡Garantía radicada!');
}

function saveCase(d){
  const casos=JSON.parse(localStorage.getItem('camiloCasos')||'[]');
  d.codigo='CV-'+String(casos.length+1).padStart(4,'0');
  casos.push(d);localStorage.setItem('camiloCasos',JSON.stringify(casos));
  return d;
}
function updateCaseStorage(){
  const casos=JSON.parse(localStorage.getItem('camiloCasos')||'[]');
  const i=casos.findIndex(c=>c.codigo===lastCase.codigo);
  if(i>-1){casos[i]=lastCase;localStorage.setItem('camiloCasos',JSON.stringify(casos));}
}

function fileToB64(f){return new Promise((res,rej)=>{const x=new FileReader();
  x.onload=()=>res(x.result.split(',')[1]);x.onerror=rej;x.readAsDataURL(f);});}
async function subirANube(caso){
  const adjuntos=[];
  if(files.serial)adjuntos.push({nombre:'foto_serial.jpg',mimeType:files.serial.type,base64:await fileToB64(files.serial)});
  if(files.novedad)adjuntos.push({nombre:'foto_novedad.jpg',mimeType:files.novedad.type,base64:await fileToB64(files.novedad)});
  if(files.video)adjuntos.push({nombre:'video_novedad.mp4',mimeType:files.video.type,base64:await fileToB64(files.video)});
  const resp=await fetch(URL_NUBE,{method:'POST',headers:{'Content-Type':'text/plain;charset=utf-8'},
    body:JSON.stringify({...caso,archivos:adjuntos})});
  const text=await resp.text();
  try{return JSON.parse(text);}catch(e){throw new Error('Respuesta no JSON: '+text.slice(0,150));}
}
async function procesarCaso(caso,titulo){
  lastCase=saveCase(caso);
  const btn=$('btnNext');
  btn.textContent='📡 Enviando a la nube…';btn.disabled=true;
  try{
    const r=await subirANube(lastCase);
    if(r&&r.ok){lastCase.nube=r.url;}else{lastCase.nube='pendiente';lastCase._err=(r&&r.error)||'desconocido';}
  }catch(e){lastCase.nube='pendiente';lastCase._err=String(e.message||e);}
  btn.disabled=false;btn.textContent='📤 Radicar caso';
  updateCaseStorage();
  $('okTitle').textContent=titulo;$('caseCode').textContent=lastCase.codigo;
  refreshCloudStatus();go('v-ok');
}
async function reintentar(){
  const btn=$('btnRetry');btn.disabled=true;btn.textContent='📡 Reintentando…';
  try{
    const r=await subirANube(lastCase);
    if(r&&r.ok){lastCase.nube=r.url;delete lastCase._err;}else{lastCase.nube='pendiente';lastCase._err=(r&&r.error)||'desconocido';}
  }catch(e){lastCase.nube='pendiente';lastCase._err=String(e.message||e);}
  btn.disabled=false;btn.textContent='📡 Reintentar envío';
  updateCaseStorage();refreshCloudStatus();
}
function refreshCloudStatus(){
  if(lastCase.nube&&lastCase.nube!=='pendiente'){
    $('cloudStatus').innerHTML='☁️ <b>Caso en la nube.</b> Camilo lo verá en su panel y en su correo.';
    $('btnRetry').style.display='none';$('debugMsg').textContent='';
  }else{
    $('cloudStatus').innerHTML='📴 <b>Sin señal por ahora:</b> el caso quedó guardado en este dispositivo.';
    $('btnRetry').style.display='block';
    if(lastCase._err)$('debugMsg').textContent='Detalle: '+lastCase._err;
  }
}
